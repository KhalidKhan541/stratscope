import os
import stratscope

execution = stratscope.start(
    api_key=os.environ["STRATSCOPE_API_KEY"],
    base_url=os.environ["STRATSCOPE_BASE_URL"],
    project_id=os.environ["STRATSCOPE_PROJECT_ID"],
    agent_id=os.environ["STRATSCOPE_AGENT_ID"],
    model="gpt-4o",
    provider="openai",
    metadata={"task": "process_refund"},
)

execution.event(
    event_type="tool_call",
    payload={"tool": "search_tickets", "query": "refund #1234"},
)

execution.finish(
    status="completed",
    latency_ms=840,
    cost_usd=0.0012,
    tokens_in=1240,
    tokens_out=212,
)

import { StratScopeClient } from "@stratscope/sdk";

// Coming to npm â€” until then, install from source:
// git clone https://github.com/KhalidKhan541/stratscope && npm install ./sdk/typescript

const client = new StratScopeClient({
  apiKey: process.env.STRATSCOPE_API_KEY,
  baseUrl: process.env.STRATSCOPE_BASE_URL,
  projectId: process.env.STRATSCOPE_PROJECT_ID,
  agentId: process.env.STRATSCOPE_AGENT_ID,
});

const exec = await client.startExecution({
  model: "gpt-4o",
  provider: "openai",
});

exec.event({
  eventType: "tool_call",
  payload: { tool: "search_tickets", query: "refund #1234" },
});

await exec.finish({
  status: "completed",
  latencyMs: 840,
  costUsd: 0.0012,
  tokensIn: 1240,
  tokensOut: 212,
});
